cmake_minimum_required(VERSION 3.20)
project(FlecsEnet VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
include(CTest)
option(FLECS_ENET_CHECK_CXX "Check C++17 public header inclusion" OFF)
option(FLECS_ENET_CUCUMBER "Register executable Cucumber features" ON)
if(FLECS_ENET_CHECK_CXX)
	enable_language(CXX)
endif()
include(cmake/dependencies.cmake)
function(flecs_enet_warnings target)
	if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
		target_compile_options(${target} PRIVATE
			-Wall -Wextra -Werror -Wpedantic)
	endif()
endfunction()
set(flecs_enet_sources src/module.c src/request.c src/transport.c
	src/events.c src/ownership.c src/reflection.c src/startup.c)
add_library(flecs_enet STATIC ${flecs_enet_sources})
add_library(FlecsEnet::flecs_enet ALIAS flecs_enet)
target_include_directories(flecs_enet PUBLIC include)
target_link_libraries(flecs_enet PUBLIC flecs_enet_flecs PRIVATE enet)
target_compile_features(flecs_enet PUBLIC c_std_11)
flecs_enet_warnings(flecs_enet)
if(BUILD_TESTING)
	add_library(flecs_enet_examples OBJECT examples/workflows.c)
	target_link_libraries(flecs_enet_examples PRIVATE
		FlecsEnet::flecs_enet enet)
	set(example_scripts "${CMAKE_CURRENT_SOURCE_DIR}/examples/scripts")
	target_compile_definitions(flecs_enet_examples PRIVATE
		FLECS_ENET_EXAMPLES="${example_scripts}")
	flecs_enet_warnings(flecs_enet_examples)
	add_subdirectory(tests)
endif()

add_executable(flecs_enet_loader examples/loader.c)
target_link_libraries(flecs_enet_loader PRIVATE FlecsEnet::flecs_enet)
flecs_enet_warnings(flecs_enet_loader)
